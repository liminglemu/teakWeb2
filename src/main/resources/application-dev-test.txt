# ============================================================
# TeakWeb2 开发环境配置文件
# 说明：此配置文件用于本地开发环境，包含所有必要的配置项
# 使用方式：通过 application.yml 中的 spring.profiles.active=dev 激活
# ============================================================

# ============================================================
# 1. 服务器配置 (Server Configuration)
# ============================================================
server:
  port: 8001                                 # 应用端口号
  address: 0.0.0.0                           # 监听地址，0.0.0.0 表示监听所有网络接口

  # Servlet 配置
  servlet:
    context-path: /                           # 应用上下文路径（访问路径前缀）
    session:
      timeout: 30m                            # Session 超时时间（分钟）

# ============================================================
# 2. 日志配置 (Logging Configuration)
# 说明：日志配置已在 logback-spring.xml 中完整定义，此处无需重复配置
#       logback-spring.xml 支持多环境（dev/test/prod），自动根据 spring.profiles.active 生效
# ============================================================
# 如需覆盖 logback-spring.xml 中的配置，可在此处添加特定配置
# 例如：临时调试某个包
# logging:
#   level:
#     com.teak.specific.package: TRACE

# ============================================================
# 3. Spring 核心配置 (Spring Core Configuration)
# ============================================================
spring:

  # 核心设置
  main:
    lazy-initialization: true                 # 启用懒加载（延迟初始化Bean，加快启动速度）

  # 应用基本信息
  application:
    name: teakWeb2                           # 应用名称（用于服务注册、监控等）

  # 文件上传配置
  servlet:
    multipart:
      max-file-size: 100MB                   # 单个文件最大大小
      max-request-size: 100MB                # 单次请求最大大小
      enabled: true                          # 启用文件上传功能

  # ============================================================
  # 4. 数据源配置 (Data Source Configuration)
  # ============================================================
  datasource:
    username: root                           # 数据库用户名
    password: Root@123456                    # 数据库密码
    driver-class-name: com.mysql.cj.jdbc.Driver  # MySQL驱动类
    # 数据库连接URL参数说明：
    # - useUnicode=true: 使用Unicode字符集
    # - characterEncoding=UTF-8: 字符编码UTF-8
    # - serverTimezone=Asia/Shanghai: 服务器时区
    # - allowPublicKeyRetrieval=true: 允许公钥检索（解决SSL错误）
    # - useSSL=false: 不使用SSL（开发环境）
    # - useInformationSchema=true: 使用information_schema库（获取表注释等）
    # - nullCatalogMeansCurrent=true: 修复MyBatis Generator问题
    url: jdbc:mysql://localhost:3306/MySqlDB?useUnicode=true&characterEncoding=UTF-8&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true&useSSL=false&useInformationSchema=true&nullCatalogMeansCurrent=true

    # HikariCP 连接池配置（高性能连接池）
    hikari:
      minimum-idle: 5                        # 最小空闲连接数
      maximum-pool-size: 20                  # 最大连接池大小
      connection-timeout: 30000              # 连接超时时间（毫秒）
      idle-timeout: 600000                   # 空闲连接超时时间（毫秒，10分钟）
      max-lifetime: 1800000                  # 连接最大生命周期（毫秒，30分钟）
      leak-detection-threshold: 5000         # 连接泄露检测阈值（毫秒）
      pool-name: TeakWeb2HikariCP            # 连接池名称（用于监控标识）
      auto-commit: true                      # 自动提交事务

  # ============================================================
  # 5. Flyway 数据库迁移配置 (Flyway Database Migration)
  # ============================================================
  flyway:
    enabled: true                            # 启用Flyway数据库版本管理
    locations: classpath:db/migration        # 迁移脚本存放路径
    baseline-on-migrate: true                # 迁移时自动创建基线
    baseline-version: 0                      # 基线版本号
    validate-on-migrate: true                # 迁移时验证脚本
    clean-disabled: true                     # 禁用clean操作（防止误删数据）

  # ============================================================
  # 6. Redis 配置 (Redis Configuration)
  # ============================================================
  data:
    redis:
      host: localhost                        # Redis服务器地址
      port: 6379                             # Redis端口
      password: Redis@123456                 # Redis密码（如果没有密码则为空）
      database: 0                            # 使用的数据库索引（0-15）
      timeout: 5000ms                        # 连接超时时间

      # Lettuce 客户端配置（Spring Boot 2.x+默认）
      lettuce:
        pool:
          max-active: 8                      # 最大活跃连接数
          max-idle: 8                        # 最大空闲连接数
          min-idle: 0                        # 最小空闲连接数
          max-wait: 2000ms                   # 获取连接最大等待时间
        shutdown-timeout: 100ms              # 关闭超时时间

  # ============================================================
  # 7. RabbitMQ 配置 (RabbitMQ Configuration)
  # 说明：队列、交换机、绑定关系已在 RabbitMQConfig.java 中通过 @Bean 配置
  #       此处仅保留基础连接配置和生产者确认配置
  # ============================================================
  rabbitmq:
    host: localhost                          # RabbitMQ服务器地址
    port: 5672                               # RabbitMQ端口（AMQP默认端口）
    username: guest                          # 用户名（默认guest）
    password: guest                          # 密码（默认guest）
    virtual-host: /                          # 虚拟主机（隔离不同应用）

    # 生产者确认机制配置（全局配置，作用于所有RabbitTemplate）
    # 注意：publisher-returns 已在 RabbitMQConfig.java 中通过 rabbitTemplate.setMandatory(true) 配置
    publisher-confirm-type: correlated        # 启用发布确认（correlated：确认到达交换机）

    # RabbitTemplate 重试配置（全局配置，作用于所有RabbitTemplate）
    template:
      retry:
        enabled: true                        # 启用重试机制
        initial-interval: 1000ms             # 初始重试间隔
        max-interval: 10000ms                # 最大重试间隔
        multiplier: 2.0                      # 重试间隔乘数（每次重试间隔倍数）

    # 注意：监听器配置已在 RabbitMQConfig.java 中通过 @Bean 定义容器工厂：
    #   - lowPrefetchFactory: prefetch=1, manual ack（适用于处理慢、需严格有序的场景）
    #   - highPrefetchFactory: prefetch=10, manual ack（适用于处理快、高吞吐场景）
    #   使用时通过 @RabbitListener(containerFactory = "xxx") 指定

  # ============================================================
  # 8. 异步任务配置 (Async Task Configuration)
  # 说明：异步线程池已在 AsyncConfig.java 中通过 @Bean 配置
  #       此处仅保留定时任务调度器配置（用于 @Scheduled）
  # ============================================================
  task:
    # 定时任务调度器配置（用于@Scheduled）
    scheduling:
      pool:
        size: 8                              # 调度线程池大小
      thread-name-prefix: teak-scheduled-    # 线程名前缀

  # ============================================================
  # 9. 缓存配置 (Cache Configuration)
  # ============================================================
  cache:
    type: CAFFEINE                            # 缓存类型（Caffeine：高性能本地缓存）
    caffeine:
      spec: maximumSize=1000,expireAfterWrite=300s  # 缓存规格（最大1000条，写入后300秒过期）

# ============================================================
# 10. MyBatis-Plus 配置 (MyBatis-Plus Configuration)
# ============================================================
mybatis-plus:
  mapper-locations: classpath*:mapper/**/*.xml   # Mapper XML文件路径（支持多模块）
  type-aliases-package: com.teak.model           # 实体类别名包路径（简化XML中的类名）

  configuration:
    log-impl: org.apache.ibatis.logging.stdout.StdOutImpl  # SQL日志实现（输出到控制台）
    # 生产环境建议使用：org.apache.ibatis.logging.nologging.NoLoggingImpl（关闭SQL日志）
    # 注意：cache-enabled 和 lazyLoadingEnabled 已在 MybatisConfig.java 中配置

    map-underscore-to-camel-case: true           # 自动下划线转驼峰命名（create_time -> createTime）
    auto-mapping-behavior: full                  # 自动映射行为（full：自动映射任意列）
    default-executor-type: reuse                # 执行器类型（reuse：重用预处理语句）
    local-cache-scope: session                  # 一级缓存作用域（session：会话级）
    jdbc-type-for-null: 'null'                  # NULL值的JDBC类型（避免空值映射错误）
    call-setters-on-nulls: true                 # 调用setter方法即使值为null（保持对象状态一致）
    return-instance-for-empty-row: true         # 空行返回空实例而非null

  global-config:
    db-config:
      # ========== ID生成策略 ==========
      # 注意：项目使用自定义 IdWorker.java 生成雪花算法ID（通过 @SnowflakeAlgorithm 注解触发）
      # 因此此处无需配置 id-type: ASSIGN_ID（MyBatis-Plus内置雪花算法）
      # 如需使用MyBatis-Plus内置雪花算法，取消下方注释并删除自定义IdWorker相关代码
      # id-type: ASSIGN_ID
      # worker-id: 1
      # datacenter-id: 1

      # ========== 逻辑删除 ==========
      # 注意：项目使用 @TableLogic 注解配置逻辑删除（见 BaseModel.java）
      # 此处无需重复配置 logic-delete-field/logic-delete-value
      # 如需全局配置逻辑删除（不推荐，灵活性差），取消下方注释并修改实体类
      # logic-delete-field: isDeleted
      # logic-delete-value: 1
      # logic-not-delete-value: 0

      # ========== 字段策略 ==========
      insert-strategy: NOT_NULL                 # 插入策略（NOT_NULL：null值不插入）
      update-strategy: NOT_NULL                 # 更新策略（NOT_NULL：null值不更新）
      # select-strategy 已在 MyBatis-Plus 3.4.0+ 移除，默认行为为 NOT_NULL
    banner: false                               # 关闭MyBatis-Plus启动Banner

# ============================================================

# 12. Spring Boot Actuator 监控配置 (Actuator Monitoring)
# ============================================================
management:
  endpoints:
    web:
      exposure:
        include: health,info,metrics,prometheus,env,beans,conditions,configprops,loggers,mappings,threaddump,heapdump  # 暴露的监控端点
    access:
      default: unrestricted
  endpoint:
    health:
      show-details: when_authorized            # 健康详情显示条件（授权用户可见）
    # loggers 端点在 Spring Boot 2.2+ 默认启用，无需显式配置 enabled
  metrics:
    enable:
      jvm:
        memory:
          used: true                          # 启用JVM内存使用指标
      system:
        cpu: true                             # 启用系统CPU使用指标
  simple:
    metrics:
      export:
        enabled: true
